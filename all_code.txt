exports.transfer = async (req, res) => {
  try {
    const { to_account_number, amount, description } = req.body;
    const transferAmount = Number(amount);
    if (!to_account_number || !transferAmount || transferAmount <= 0) {
      return res.status(400).json({ message: 'Sá»‘ tiá»n hoáº·c tÃ i khoáº£n nháº­n khÃ´ng há»£p lá»‡!' });
    }

    // 1. TÃ¬m tÃ i khoáº£n ngÆ°á»i gá»­i (tá»« JWT token)
    const fromAccount = await Account.findOne({ user_id: req.user.id });
    if (!fromAccount) {
      return res.status(404).json({ message: 'TÃ i khoáº£n ngÆ°á»i gá»­i khÃ´ng tá»“n táº¡i!' });
    }

    // 2. TÃ¬m tÃ i khoáº£n ngÆ°á»i nháº­n
    const toAccount = await Account.findOne({ account_number: to_account_number });
    if (!toAccount) {
      return res.status(404).json({ message: 'TÃ i khoáº£n ngÆ°á»i nháº­n khÃ´ng tá»“n táº¡i!' });
    }

    // 3. Kiá»ƒm tra chuyá»ƒn tiá»n cho chÃ­nh mÃ¬nh
    if (fromAccount.account_number === toAccount.account_number) {
      return res.status(400).json({ message: 'KhÃ´ng thá»ƒ tá»± chuyá»ƒn tiá»n cho chÃ­nh mÃ¬nh!' });
    }

    // 4. Kiá»ƒm tra sá»‘ dÆ°
    if (fromAccount.balance < transferAmount) {
      return res.status(400).json({ message: 'Sá»‘ dÆ° khÃ´ng Ä‘á»§ Ä‘á»ƒ thá»±c hiá»‡n giao dá»‹ch!' });
    }

    // 5. Trá»« tiá»n ngÆ°á»i gá»­i vÃ  cá»™ng tiá»n ngÆ°á»i nháº­n
    fromAccount.balance -= transferAmount;
    toAccount.balance += transferAmount;

    await fromAccount.save();
    await toAccount.save();

    // 6. Ghi nháº­t kÃ½ giao dá»‹ch
    const newTransaction = new Transaction({
      from_account_id: fromAccount._id,
      to_account_id: toAccount._id,
      amount: transferAmount,
      description: description || 'Chuyá»ƒn tiá»n ná»™i bá»™',
      status: 'success',
    });
    await newTransaction.save();

    return res.status(200).json({
      message: 'Chuyá»ƒn tiá»n thÃ nh cÃ´ng!',
      remaining_balance: fromAccount.balance,
      transaction: newTransaction,
    });
  } catch (error) {
    console.error('Transfer error:', error);
    return res.status(500).json({ message: 'Lá»—i server khi chuyá»ƒn tiá»n!' });
  }
};

      return res.status(404).json({ message: 'KhÃ´ng tÃ¬m tháº¥y tÃ i khoáº£n!' });

      $or: [{ from_account_id: account._id }, { to_account_id: account._id }],

        path: 'to_account_id',
        select: 'account_number',
        populate: { path: 'user_id', select: 'full_name' },
      })

  try {
    const { email, password } = req.body;

    if (!email || !password) {
      return res.status(400).json({ message: 'Vui lÃ²ng nháº­p email vÃ  máº­t kháº©u!' });
    }

    const user = await User.findOne({ email });
    if (!user) {
      return res.status(400).json({ message: 'Email hoáº·c máº­t kháº©u khÃ´ng chÃ­nh xÃ¡c!' });

      user: {

const jwt = require('jsonwebtoken');

const authMiddleware = (req, res, next) => {
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ message: 'KhÃ´ng cÃ³ quyá»n truy cáº­p, vui lÃ²ng Ä‘Äƒng nháº­p!' });
  }

  const token = authHeader.split(' ')[1];

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET || 'banking_secret_key_123');
    req.user = decoded; // { id, role }
    next();
  } catch (error) {
    return res.status(401).json({ message: 'Token khÃ´ng há»£p lá»‡ hoáº·c Ä‘Ã£ háº¿t háº¡n!' });
  }
};

module.exports = authMiddleware;
